get_eia_sep <- function(state, abbr){
    library(tidyverse)
    library(stringr)
    library(httr2)
    library(rvest)
    library(fs)
  
    state_formatted <- str_to_lower(state) |> str_replace_all("\\s", "")
    
    dir_name <- fs::path("data", "mp02")
    file_name <- fs::path(dir_name, state_formatted)
    
    fs::dir_create(dir_name, recurse=TRUE)
    
    if(!fs::file_exists(file_name)){
        BASE_URL <- "https://www.eia.gov"
        RESPONSE <- request(BASE_URL) |> 
          req_url_path("electricity", 
                       "state", 
                       state_formatted) |>
          req_headers(
            `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0"
            ) |>   
          req_retry(
                max_tries = 4, 
          ) |>
          req_perform(file_name)
    
        resp_check_status(RESPONSE)
    }
    
    TABLE <- read_html(file_name) |> 
        html_element("table") |> 
        html_table() |>
        mutate(Item = str_to_lower(Item))
    
    if("U.S. rank" %in% colnames(TABLE)){
        TABLE <- TABLE |> rename(Rank = `U.S. rank`)
    }
    
    CO2_MWh <- TABLE |> 
        filter(Item == "carbon dioxide (lbs/mwh)") |>
        pull(Value) |> 
        str_replace_all(",", "") |>
        as.numeric()
    
    PRIMARY <- TABLE |> 
        filter(Item == "primary energy source") |> 
        pull(Rank)
    
    RATE <- TABLE |>
        filter(Item == "average retail price (cents/kwh)") |>
        pull(Value) |>
        as.numeric()
    
    GENERATION_MWh <- TABLE |>
        filter(Item == "net generation (megawatthours)") |>
        pull(Value) |>
        str_replace_all(",", "") |>
        as.numeric()
    
    data.frame(CO2_MWh               = CO2_MWh, 
               primary_source        = PRIMARY,
               electricity_price_MWh = RATE * 10,
               generation_MWh        = GENERATION_MWh, 
               state                 = state, 
               abbreviation          = abbr
    )
}

load_ntd_energy <- function(){
  library(readxl)
  library(httr2)
  library(tidyverse)
  library(fs)

  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse = TRUE)

  NTD_ENERGY_FILE <- fs::path(DATA_DIR, "2025_ntd_energy.xlsx")

  if(!fs::file_exists(NTD_ENERGY_FILE)){
    request("https://www.transit.dot.gov/") |>
      req_url_path(
        "sites",
        "fta.dot.gov",
        "files",
        "2026-09",
        "2025 Energy Consumption_260806.xlsx"
      ) |>
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0"
      ) |>
      req_perform(NTD_ENERGY_FILE)
  }

  if(file.info(NTD_ENERGY_FILE)$size < 1e4){
    stop("Download failed. Delete the corrupted file and try again.")
  }

  to_numeric_fill_0 <- function(x){
    x <- if_else(x == "-", NA, x)
    replace_na(as.numeric(x), 0)
  }

  read_excel(NTD_ENERGY_FILE) |>
    select(-c(
      `Reporter Type`,
      `Reporting Module`,
      `Gallons of Other Fuel`,
      `Other Fuel Description`
    )) |>
    mutate(across(
      -c(`Agency Name`, `Mode`, `TOS`),
      to_numeric_fill_0
    )) |>
    group_by(`NTD ID`, `Mode`, `Agency Name`) |>
    summarize(
      across(where(is.numeric), sum),
      .groups = "keep"
    ) |>
    mutate(
      ENERGY = sum(c_across(where(is.numeric)))
    ) |>
    filter(ENERGY > 0) |>
    select(-ENERGY) |>
    ungroup()
}

NTD_ENERGY <- load_ntd_energy()

load_ntd_service <- function(){
  library(httr2)
  library(tidyverse)
  library(fs)

  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse = TRUE)

  NTD_SERVICE_FILE <- fs::path(DATA_DIR, "2025_ntd_service.csv")

  if(!fs::file_exists(NTD_SERVICE_FILE)){
    request("https://data.transportation.gov/resource/4fir-qbim.csv") |>
      req_url_query(`$query` = paste(
        "SELECT `_5_digit_ntd_id` AS `NTD_ID`",
        "`max_agency` AS `AGENCY`",
        "`max_reporter_type` AS `reporter_type`",
        "`max_organization_type` AS `organization_type`",
        "`max_city` AS `city`",
        "`max_state` AS `state`",
        "`mode`",
        "`type_of_service`",
        "`max_primary_uza_code` as `uza_code`",
        "`max_primary_uza_name` as `uza_name`",
        "`max_primary_uza_area_sq_miles` as `uza_sq_miles`",
        "`max_primary_uza_population` as `uza_population`",
        "`max_service_area_sq_miles` as `service_area_sq_miles`",
        "`max_service_area_population` as `service_area_population`",
        "`questionable_record` as `flagged_likely_inaccurate`",
        "`sum_unlinked_passenger_trips_upt` as `total_upt`",
        "`sum_passenger_miles` WHERE caseless_one_of(`report_year`, '2025') limit 50000",
        sep = ","
      )) |>
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0"
      ) |>
      req_perform(NTD_SERVICE_FILE)
  }

  if(file.info(NTD_SERVICE_FILE)$size < 1e4){
    stop("Download failed. Please check the cached service file.")
  }

  read_csv(NTD_SERVICE_FILE) |>
    mutate(
      flagged_likely_inaccurate = !is.na(flagged_likely_inaccurate)
    )
}

NTD_SERVICE <- load_ntd_service()

library(dplyr)

# Conversion factors: pounds of CO2 per reported unit
# Conventional fuels: EIA
# Biodiesel, ethanol, LPG, LNG: EPA
# CNG: estimated using a gasoline gallon equivalent
# Hydrogen: direct CO2 only; production not included

kg_to_lb <- 2.20462262

diesel_factor <- 22.45
gasoline_factor <- 20.86
cng_factor <- 116.65 * 0.114
biodiesel_factor <- 9.45 * kg_to_lb
ethanol_factor <- 5.75 * kg_to_lb
lpg_factor <- 5.68 * kg_to_lb
lng_factor <- 4.50 * kg_to_lb
kerosene_factor <- 21.78
bunker_factor <- 24.78

# Calculate emissions for every agency-mode pair
transit_emissions <- transit_combined |>
  mutate(
    fuel_co2_lbs =
      `Gallons of Diesel Fuel` * diesel_factor +
      `Gallons of Gasoline` * gasoline_factor +
      `Gallons of Compressed Natural Gas` * cng_factor +
      `Gallons of Bio-Diesel` * biodiesel_factor +
      `Gallons of Ethanol` * ethanol_factor +
      `Gallons of Liquified Petroleum Gas` * lpg_factor +
      `Gallons of Liquified Nat Gas` * lng_factor +
      `Gallons of Kerosene` * kerosene_factor +
      `Gallons of Bunker Fuel` * bunker_factor,

    total_electric_kwh =
      `Kilowatt hours of Electric Propulsion` +
      `Kilowatt hours of Electric Battery`,

    electric_co2_lbs = case_when(
      total_electric_kwh == 0 ~ 0,
      !is.na(CO2_MWh) ~
        (total_electric_kwh / 1000) * CO2_MWh,
      TRUE ~ NA_real_
    ),

    total_co2_lbs = fuel_co2_lbs + electric_co2_lbs,

    total_co2_tons = total_co2_lbs / 2000
  )

# Check calculation results

    calculated_records = sum(!is.na(total_co2_lbs)),
    missing_emissions = sum(is.na(total_co2_lbs)),

Press Release

The Green Transit Alliance is announcing its first Transit Efficiency Awards to recognize public transit agencies that move people while producing very little carbon pollution. Ridership alone doesn’t tell the full story, and neither do emissions alone. Looking at them together shows how much climate impact each rider is responsible for. This year’s winners, the City of Seattle’s monorail service and Oregon’s TriMet, stood out because their emissions per rider were a small fraction of what is typical for agencies like them.

Award 1 — City of Seattle (Monorail)

The City of Seattle’s reported monorail service earned the award for the lowest estimated emissions per passenger mile among agencies reporting at least 100,000 trips. The monorail produced just 0.064 lbs of CO₂ per passenger mile. The median for the 484 eligible agencies was 1.107 lbs, so Seattle’s figure is about 94% lower, or roughly one-seventeenth of the typical agency. This result applies only to the monorail operation Seattle reported, not to the city’s wider transit network.

Award 2 — TriMet (Oregon)

The Tri-County Metropolitan Transportation District of Oregon (TriMet) earned the award for large agencies serving at least 10 million passenger trips a year. TriMet produced an estimated 0.413 lbs of CO₂ per passenger trip, while the median among the 74 eligible agencies was 3.267 lbs. Doing that at this scale shows that a big, heavily used system can still be very efficient.

Closing and Limitations

These estimates combine 2025 National Transit Database (NTD) ridership and fuel data with 2024 U.S. Energy Information Administration (EIA) electricity data, using standard emissions conversion assumptions. One Puerto Rico agency was left out of the complete-emissions rankings because no electricity emissions factor was available for it. The figures should be read as careful estimates, not exact measurements.
